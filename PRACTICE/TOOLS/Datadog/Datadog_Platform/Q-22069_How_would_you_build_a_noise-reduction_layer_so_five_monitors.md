# Example Feature Engineering Code
import pandas as pd

def extract_features(alerts):
    features = alerts.groupby(['service', 'error_message']) \
                    .agg({'timestamp': ['count', 'min', 'max'],
                          'severity': 'mean'}) \
                    .reset_index()
    return features
```
